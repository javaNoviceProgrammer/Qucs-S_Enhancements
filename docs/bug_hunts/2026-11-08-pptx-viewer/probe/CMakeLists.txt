cmake_minimum_required(VERSION 3.21)
project(pptx_probe CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC OFF)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg)
find_package(ZLIB REQUIRED)
set(SRC "${QUCS_SRC}")
add_executable(pptx_probe pptx_probe.cpp ${SRC}/pptx.cpp ${SRC}/zipfile.cpp ${SRC}/svgguard.cpp)
target_include_directories(pptx_probe PRIVATE ${SRC})
target_link_libraries(pptx_probe Qt6::Core Qt6::Gui Qt6::Svg ZLIB::ZLIB)
